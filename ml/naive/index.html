type Feature = String
type Class = String
type Dataset = [([Feature], Class)]

train :: Dataset -> (Map Class Double, Map (Feature, Class) Double)
train data = (priors, likelihoods)
  where
    n = fromIntegral (length data)
    classes = nub (map snd data)

    priors = Map.fromList
      [(c, count c / n) | c <- classes]
      where count c = fromIntegral . length $ filter ((==c) . snd) data

    likelihoods = Map.fromList
      [((f, c), prob f c) | c <- classes, f <- allFeatures]

    prob f c = (countFC + 1) / (countC + vocab)
      where
        countFC = fromIntegral . length $ filter (hasFeature f c) data
        countC  = fromIntegral . length $ filter ((==c) . snd) data
        vocab   = fromIntegral (length allFeatures)

predict :: (Map Class Double, Map (Feature, Class) Double)
        -> [Feature] -> Class
predict (priors, likes) features =
  fst . maximumBy (comparing snd) $
    [(c, score c) | c <- Map.keys priors]
  where
    score c = log (priors ! c) +
              sum [log (likes ! (f, c)) | f <- features]
