K-NEAREST NEIGHBORS

type Point = [Double]
type Label = String
type Dataset = [(Point, Label)]

distance :: Point -> Point -> Double
distance a b = sqrt . sum $ zipWith (\x y -> (x-y)^2) a b

knn :: Int -> Dataset -> Point -> Label
knn k dataset query = majority . map snd . take k . sortOn fst $
  [(distance query p, l) | (p, l) <- dataset]
  where
    majority = head . maximumBy (comparing length) . group . sort

EXAMPLE

dataset = [([0,0], "A"), ([1,1], "A"), ([5,5], "B"), ([6,6], "B")]

-- Classify point [2,2] with k=3
result = knn 3 dataset [2,2]  -- "A"
