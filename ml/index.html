FUNCTIONAL ML PATTERNS

type Model a b = a -> b
type Train a b = [(a,b)] -> Model a b
type Loss b = b -> b -> Double

fit :: Train a b -> [(a,b)] -> Model a b
predict :: Model a b -> a -> b
score :: Loss b -> Model a b -> [(a,b)] -> Double
