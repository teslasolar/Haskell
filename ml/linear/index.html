LINEAR REGRESSION

type Point = (Double, Double)

linearFit :: [Point] -> (Double, Double)  -- (slope, intercept)
linearFit pts = (m, b)
  where
    n  = fromIntegral (length pts)
    xs = map fst pts
    ys = map snd pts
    sumX  = sum xs
    sumY  = sum ys
    sumXY = sum (zipWith (*) xs ys)
    sumX2 = sum (map (^2) xs)
    m = (n * sumXY - sumX * sumY) / (n * sumX2 - sumX^2)
    b = (sumY - m * sumX) / n

PREDICT

predict :: (Double, Double) -> Double -> Double
predict (m, b) x = m * x + b

-- Example
model = linearFit [(1,2), (2,4), (3,6)]
result = predict model 5  -- 10.0
