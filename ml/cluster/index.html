K-MEANS

type Point = [Double]
type Centroid = Point

kmeans :: Int -> [Point] -> [Centroid]
kmeans k points = iterate step initial !! 100
  where
    initial = take k points

    step centroids = map centroid (assign centroids)

    assign cs = groupBy ((==) `on` nearest cs) $
                sortOn (nearest cs) points

    nearest cs p = minimumBy (comparing (distance p)) cs

    centroid ps = map (/ n) $ foldl1 (zipWith (+)) ps
      where n = fromIntegral (length ps)

distance :: Point -> Point -> Double
distance a b = sqrt . sum $ zipWith (\x y -> (x-y)^2) a b
