A* SEARCH

astar :: (Ord a) => a -> (a -> Bool)
      -> (a -> [a]) -> (a -> Int) -> Maybe [a]
astar start goal neighbors h = go (singleton (h start, [start])) Set.empty
  where
    go pq visited
      | null pq = Nothing
      | goal current = Just (reverse path)
      | current `Set.member` visited = go rest visited
      | otherwise = go pq' (Set.insert current visited)
      where
        ((_, path@(current:_)), rest) = deleteFindMin pq
        pq' = foldr (insert . score) rest (neighbors current)
        score n = (length path + h n, n : path)

BFS / DFS

bfs :: Ord a => a -> (a -> [a]) -> [a]
bfs start neighbors = go (Seq.singleton start) Set.empty
  where
    go queue visited
      | Seq.null queue = []
      | x `Set.member` visited = go xs visited
      | otherwise = x : go (xs <> Seq.fromList (neighbors x))
                           (Set.insert x visited)
      where (x :< xs) = Seq.viewl queue

dfs :: Ord a => a -> (a -> [a]) -> [a]
dfs start neighbors = go [start] Set.empty
  where
    go [] _ = []
    go (x:xs) visited
      | x `Set.member` visited = go xs visited
      | otherwise = x : go (neighbors x ++ xs) (Set.insert x visited)
