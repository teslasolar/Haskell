CHROMOSOME

type Gene = Bool
type Chromosome = [Gene]
type Population = [Chromosome]

fitness :: Chromosome -> Double
fitness = fromIntegral . length . filter id

SELECTION & CROSSOVER

select :: Population -> IO (Chromosome, Chromosome)
select pop = do
  let weighted = [(fitness c, c) | c <- pop]
  a <- roulette weighted
  b <- roulette weighted
  return (a, b)

crossover :: Chromosome -> Chromosome -> IO Chromosome
crossover a b = do
  point <- randomRIO (0, length a - 1)
  return $ take point a ++ drop point b

mutate :: Double -> Chromosome -> IO Chromosome
mutate rate = mapM $ \g -> do
  r <- randomRIO (0, 1)
  return $ if r < rate then not g else g

EVOLUTION

evolve :: Int -> Population -> IO Population
evolve 0 pop = return pop
evolve n pop = do
  offspring <- replicateM (length pop) $ do
    (a, b) <- select pop
    child  <- crossover a b
    mutate 0.01 child
  evolve (n-1) offspring
