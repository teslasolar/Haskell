NETWORK

type Layer = [[Double]]  -- weights matrix
type Network = [Layer]

sigmoid :: Double -> Double
sigmoid x = 1 / (1 + exp (-x))

relu :: Double -> Double
relu = max 0

FORWARD PASS

forward :: Network -> [Double] -> [Double]
forward net input = foldl layer input net
  where
    layer xs weights = map (sigmoid . sum . zipWith (*) xs) weights

predict :: Network -> [Double] -> Int
predict net = argmax . forward net

BACKPROP

backprop :: Network -> [Double] -> [Double] -> Network
backprop net input target = zipWith updateLayer net deltas
  where
    outputs = scanl (flip layer) input net
    errors  = zipWith (-) target (last outputs)
    deltas  = scanr propError errors (zip net (init outputs))

    propError (w, o) nextDelta =
      [sum (zipWith (*) col nextDelta) * o' * (1-o')
      | (col, o') <- zip (transpose w) o]

    updateLayer weights delta =
      [[w - 0.1 * d * o | (w, o) <- zip row outputs]
      | (row, d) <- zip weights delta]
