Q-LEARNING

-- Q-Learning in Haskell
type State = Int
type Action = Int
type QTable = Map (State, Action) Double

qUpdate :: Double -> Double -> QTable
        -> State -> Action -> Double -> State -> QTable
qUpdate α γ q s a r s' =
  let maxQ = maximum [q ! (s', a') | a' <- actions]
      old  = q ! (s, a)
      new  = old + α * (r + γ * maxQ - old)
  in insert (s, a) new q

POLICY

εGreedy :: Double -> QTable -> State -> IO Action
εGreedy ε q s = do
  r <- randomRIO (0, 1)
  if r < ε
    then randomChoice actions
    else return $ argmax (\a -> q ! (s, a)) actions

TRAINING LOOP

train :: Int -> QTable -> IO QTable
train 0 q = return q
train n q = do
  (s, a, r, s') <- step env (εGreedy 0.1 q)
  let q' = qUpdate 0.1 0.99 q s a r s'
  train (n-1) q'
