minimax :: Player -> GameState -> Int
minimax _ state | terminal state = evaluate state
minimax Max state = maximum [minimax Min s | s <- moves state]
minimax Min state = minimum [minimax Max s | s <- moves state]

ALPHA-BETA PRUNING

alphabeta :: Player -> Int -> Int -> GameState -> Int
alphabeta _ α β state | terminal state = evaluate state
alphabeta Max α β state = go α (moves state)
  where
    go α [] = α
    go α (s:ss)
      | α >= β = α
      | otherwise = go (max α (alphabeta Min α β s)) ss
alphabeta Min α β state = go β (moves state)
  where
    go β [] = β
    go β (s:ss)
      | α >= β = β
      | otherwise = go (min β (alphabeta Max α β s)) ss
