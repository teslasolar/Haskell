UDT:HOF
map :: (a->b) -> [a] -> [b]
map (+1) [1,2,3] = [2,3,4]

filter :: (a->Bool) -> [a] -> [a]
filter (>2) [1,2,3,4] = [3,4]

foldr :: (a->b->b) -> b -> [a] -> b
foldr (+) 0 [1,2,3] = 6

UDT:Sections
(+1) = \x->x+1
(*2) = \x->x*2

UDT:Composition
(.) :: (b->c) -> (a->b) -> a -> c
(f . g) x = f (g x)
