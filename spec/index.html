HEADER UDT SPECIFICATION

Every file starts with header block
Logical time, not system clock

UDT:Header = {
  uid: UUID,        # unique file id
  v:   Nat,         # version (monotonic)
  t:   LTime,       # logical timestamp
  p:   [UUID],      # parent refs
  h:   Hash         # content hash
}

UDT:LTime = {
  seq: Nat,         # sequence number
  src: NodeId       # origin node
}

# Distance = |t1.seq - t2.seq|
# Incompatible if distance > threshold

HEADER FORMAT

# In JS/MD files:
/*! K:uid:v:seq:src:hash !*/

# In HTML:
<!-- K:uid:v:seq:src:hash -->

# Example:
/*! K:a1b2c3:3:42:node0:x7f8 !*/
